 4pub struct CriticallyDampedSpringAnimation {
 5    pub position: f32,
 6    velocity: f32,
 7}
 8
 9impl CriticallyDampedSpringAnimation {
10    pub fn new() -> Self {
11        Self {
12            position: 0.0,
13            velocity: 0.0,
14        }
15    }
16
17    pub fn update(&mut self, dt: f32, animation_length: f32) -> bool {
18        if animation_length <= dt {
19            self.reset();
20            return false;
21        }
22
23        if self.position == 0.0 {
24            return false;
25        }
26
27        // Simulate a critically damped spring, also known as a PD controller.
28        // For more details of why this was chosen, see this:
29        // https://gdcvault.com/play/1027059/Math-In-Game-Development-Summit
30        // < 1 underdamped,  1 critically damped, > 1 overdamped
31        let zeta = 1.0;
32        // The omega is calculated so that the destination is reached with a 2% tolerance in
33        // animation_length time.
34        let omega = 4.0 / (zeta * animation_length);
35
36        // Use the analytica formula for critically damped harmonic oscillation
37        // a and b are the intial conditions by setting dt to zero and solving the position and
38        // velocity respectively
39        let a = self.position;
40        let b = self.position * omega + self.velocity;
41        let c = (-omega * dt).exp();
42        self.position = (a + b * dt) * c;
43        self.velocity = c * (-a * omega - b * dt * omega + b);
44        if self.position.abs() < 0.01 {
45            self.reset();
46            false
47        } else {
48            true
49        }
50
51    }
52
53    pub fn reset(&mut self) {
54        self.position = 0.0;
55        self.velocity = 0.0;
56
57    }

 1export class SmoothScrollingOperation {
 2
 3	public readonly from: ISmoothScrollPosition;
 4	public to: ISmoothScrollPosition;
 5	public readonly duration: number;
 6	public readonly startTime: number;
 7	public animationFrameDisposable: IDisposable | null;
 8
 9	private scrollLeft!: IAnimation;
10	private scrollTop!: IAnimation;
11
12	constructor(from: ISmoothScrollPosition, to: ISmoothScrollPosition, startTime: number, duration: number) {
13		this.from = from;
14		this.to = to;
15		this.duration = duration;
16		this.startTime = startTime;
17
18		this.animationFrameDisposable = null;
19
20		this._initAnimations();
21	}
22
23	private _initAnimations(): void {
24		this.scrollLeft = this._initAnimation(this.from.scrollLeft, this.to.scrollLeft, this.to.width);
25		this.scrollTop = this._initAnimation(this.from.scrollTop, this.to.scrollTop, this.to.height);
26	}
27
28	private _initAnimation(from: number, to: number, viewportSize: number): IAnimation {
29		const delta = Math.abs(from - to);
30		if (delta > 2.5 * viewportSize) {
31			let stop1: number, stop2: number;
32			if (from < to) {
33				// scroll to 75% of the viewportSize
34				stop1 = from + 0.75 * viewportSize;
35				stop2 = to - 0.75 * viewportSize;
36			} else {
37				stop1 = from - 0.75 * viewportSize;
38				stop2 = to + 0.75 * viewportSize;
39			}
40			return createComposed(createEaseOutCubic(from, stop1), createEaseOutCubic(stop2, to), 0.33);
41		}
42		return createEaseOutCubic(from, to);
43	}
44
45	public dispose(): void {
46		if (this.animationFrameDisposable !== null) {
47			this.animationFrameDisposable.dispose();
48			this.animationFrameDisposable = null;
49		}
50	}
51
52	public acceptScrollDimensions(state: ScrollState): void {
53		this.to = state.withScrollPosition(this.to);
54		this._initAnimations();
55	}
56
57	public tick(): SmoothScrollingUpdate {
58		return this._tick(Date.now());
59	}
60
61	protected _tick(now: number): SmoothScrollingUpdate {
62		const completion = (now - this.startTime) / this.duration;
63
64		if (completion < 1) {
65			const newScrollLeft = this.scrollLeft(completion);
66			const newScrollTop = this.scrollTop(completion);
67			return new SmoothScrollingUpdate(newScrollLeft, newScrollTop, false);
68		}
69
70		return new SmoothScrollingUpdate(this.to.scrollLeft, this.to.scrollTop, true);
71	}
72
73	public combine(from: ISmoothScrollPosition, to: ISmoothScrollPosition, duration: number): SmoothScrollingOperation {
74		return SmoothScrollingOperation.start(from, to, duration);
75	}
76
77	public static start(from: ISmoothScrollPosition, to: ISmoothScrollPosition, duration: number): SmoothScrollingOperation {
78		// +10 / -10 : pretend the animation already started for a quicker response to a scroll request
79		duration = duration + 10;
80		const startTime = Date.now() - 10;
81
82		return new SmoothScrollingOperation(from, to, startTime, duration);
83	}
84}
